# VHOL Demo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a fully-live, two-view demo (Acme Co store + devtools-style monitoring console) of an external customer-support agent running on Merge Gateway + Agent Handler.

**Architecture:** Single Vite + React + TS app. The browser holds two views (`/` store, `/console` monitor). A Vite dev-server middleware holds all Merge keys server-side, runs the agentic loop (port of the existing Python), manages the Registered User (RU) lifecycle, and broadcasts structured `AgentEvent`s to the console over SSE.

**Tech Stack:** Vite 6, React 18, TypeScript, Tailwind v4, Vitest + @testing-library/react (jsdom), native `fetch`, Server-Sent Events.

**Reference:** Spec at `docs/superpowers/specs/2026-06-01-vhol-demo-design.md`. Source of truth for the loop: `~/Desktop/AH + Gateway/{gateway,agent_handler,main}.py`. Source of truth for middleware/key-handling pattern: `~/Desktop/ACT_DemoLink/{vite.config.ts,src/server/createLinkToken.ts}`.

```ts
export const EVENT_SOURCES = [
  "identity",
  "gateway",
  "agent_handler",
  "dlp",
  "response",
  "error",
] as const;

export type EventSource = (typeof EVENT_SOURCES)[number];

export interface AgentEvent {
  id: string;
  ts: number;
  source: EventSource;
  kind: string; // ru_lookup | ru_create | ru_exists | route | tool_search | tool_call | tool_result | redaction | message | login | error
  summary: string;
  detail?: unknown;
}

export interface SessionInfo {
  ruId: string;
  created: boolean;
  name: string;
  company: string;
  originUserId: string;
}

export function isAgentEvent(value: unknown): value is AgentEvent {
  if (typeof value !== "object" || value === null) return false;
  const e = value as Record<string, unknown>;
  return (
    typeof e.id === "string" &&
    typeof e.ts === "number" &&
    typeof e.kind === "string" &&
    typeof e.summary === "string" &&
    typeof e.source === "string" &&
    (EVENT_SOURCES as readonly string[]).includes(e.source)
  );
}
```

```ts
import type { IncomingMessage, ServerResponse } from "node:http";

export function readBody(req: IncomingMessage): Promise<string> {
  return new Promise((resolve, reject) => {
    let data = "";
    req.on("data", (chunk: Buffer) => (data += chunk));
    req.on("end", () => resolve(data));
    req.on("error", reject);
  });
}

export async function readJson<T = Record<string, unknown>>(req: IncomingMessage): Promise<T> {
  const raw = await readBody(req);
  return (raw ? JSON.parse(raw) : {}) as T;
}

export function sendJson(res: ServerResponse, status: number, body: unknown): void {
  res.statusCode = status;
  res.setHeader("Content-Type", "application/json");
  res.end(JSON.stringify(body));
}
```

```ts
import type { Env, GatewayResponse, HistoryItem, Tool } from "./types";

const GATEWAY_URL = "https://api-gateway.merge.dev/v1/responses";

export async function chat(
  env: Env,
  model: string,
  history: HistoryItem[],
  tools: Tool[]
): Promise<GatewayResponse> {
  const res = await fetch(GATEWAY_URL, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${env.GATEWAY_API_KEY}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      ...(model ? { model } : {}),
      input: history,
      ...(tools.length ? { tools, tool_choice: "auto" } : {}),
      project_id: env.PROJECT_ID,
    }),
  });
  if (!res.ok) {
    throw new Error(`Gateway ${res.status} ${res.statusText}: ${await res.text()}`);
  }
  return (await res.json()) as GatewayResponse;
}
```

```ts
import type { Env, Tool } from "./types";

const MAX_INTENT_LEN = 300;

function execBaseUrl(env: Env, ruId: string): string {
  return `https://ah-api.merge.dev/api/v1/tool-packs/${env.TOOL_PACK_ID}/registered-users/${ruId}`;
}

function authHeaders(env: Env): Record<string, string> {
  return { Authorization: `Bearer ${env.AH_API_KEY}`, "Content-Type": "application/json" };
}

export async function searchTools(
  env: Env,
  ruId: string,
  intent: string,
  maxResults = 5
): Promise<Tool[]> {
  const res = await fetch(`${execBaseUrl(env, ruId)}/search/`, {
    method: "POST",
    headers: authHeaders(env),
    body: JSON.stringify({ intent: intent.slice(0, MAX_INTENT_LEN), max_results: maxResults }),
  });
  if (!res.ok) return []; // proceed without tools, like the python client
  const data = (await res.json()) as {
    tools?: { fully_qualified_name: string; description: string; input_schema: unknown }[];
  };
  return (data.tools ?? []).map((t) => ({
    type: "function",
    name: t.fully_qualified_name,
    description: t.description,
    parameters: t.input_schema,
  }));
}

export async function callTool(
  env: Env,
  ruId: string,
  name: string,
  args: Record<string, unknown>
): Promise<string> {
  const res = await fetch(`${execBaseUrl(env, ruId)}/mcp/`, {
    method: "POST",
    headers: authHeaders(env),
    body: JSON.stringify({ jsonrpc: "2.0", method: "tools/call", id: 1, params: { name, arguments: args } }),
  });
  if (!res.ok) {
    throw new Error(`Agent Handler ${res.status} ${res.statusText}: ${await res.text()}`);
  }
  const data = (await res.json()) as { result: { content: { text: string }[] } };
  return data.result.content[0].text;
}
```

```ts
const REDACTION_MARKERS = ["[REDACTED]", "***REDACTED***", "<redacted>"];
const SSN_PATTERN = /\b\d{3}-\d{2}-\d{4}\b/;

/** Returns human-readable labels for any DLP signals found in a tool result. */
export function detectRedactions(text: string): string[] {
  const hits: string[] = [];
  for (const marker of REDACTION_MARKERS) {
    if (text.includes(marker)) hits.push(marker);
  }
  if (SSN_PATTERN.test(text)) hits.push("SSN pattern");
  return hits;
}
```

```ts
import type { AgentEvent, EventSource } from "../lib/events";

type Subscriber = (event: AgentEvent) => void;

class EventBus {
  private log: AgentEvent[] = [];
  private subscribers = new Set<Subscriber>();
  private seq = 0;

  emit(source: EventSource, kind: string, summary: string, detail?: unknown): AgentEvent {
    const event: AgentEvent = {
      id: `evt_${++this.seq}`,
      ts: Date.now(),
      source,
      kind,
      summary,
      detail,
    };
    this.log.push(event);
    for (const fn of this.subscribers) fn(event);
    return event;
  }

  subscribe(fn: Subscriber): () => void {
    this.subscribers.add(fn);
    return () => this.subscribers.delete(fn);
  }

  getLog(): AgentEvent[] {
    return [...this.log];
  }

  clear(): void {
    this.log = [];
  }
}

export const bus = new EventBus();
```

```ts
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { dirname, join } from "node:path";
import { bus } from "./eventBus";
import * as gateway from "./gateway";
import * as ah from "./agentHandler";
import { detectRedactions } from "./dlp";
import type { Env, GatewayContentItem, HistoryItem } from "./types";

export interface Session {
  ruId: string | null;
  originUserId: string;
  name: string;
  company: string;
  companyId: string;
  history: HistoryItem[];
  model: string;
}

function systemPrompt(): string {
  const here = dirname(fileURLToPath(import.meta.url));
  return readFileSync(join(here, "systemPrompt.md"), "utf8");
}

function systemMessage(): HistoryItem {
  return { type: "message", role: "system", content: systemPrompt() };
}

export function createSession(env: Env): Session {
  return {
    ruId: null,
    originUserId: env.DEMO_USER_ORIGIN_ID,
    name: env.DEMO_USER_NAME,
    company: env.DEMO_COMPANY_NAME,
    companyId: env.DEMO_COMPANY_ID,
    history: [systemMessage()],
    model: "",
  };
}

export async function runTurn(env: Env, session: Session, userMessage: string): Promise<string> {
  if (!session.ruId) throw new Error("Session has no Registered User. Call startSession first.");

  session.history.push({ type: "message", role: "user", content: userMessage });

  const tools = await ah.searchTools(env, session.ruId, userMessage);
  bus.emit(
    "agent_handler",
    "tool_search",
    tools.length ? `found ${tools.length} tool(s)` : "no tools needed",
    { intent: userMessage, tools: tools.map((t) => t.name) }
  );

  while (true) {
    const response = await gateway.chat(env, session.model, session.history, tools);
    const returnedModel = response.model ?? "";
    if (returnedModel.includes("/")) session.model = returnedModel;
    bus.emit("gateway", "route", `routed → ${returnedModel || "unknown"}`, {
      model: returnedModel,
      usage: response.usage,
    });

    const output = response.output[0];
    session.history.push(output as HistoryItem);

    if (output.finish_reason === "tool_use") {
      for (const item of output.content as GatewayContentItem[]) {
        if (item.type !== "tool_use") continue;
        bus.emit("agent_handler", "tool_call", `${item.name}(…)`, {
          name: item.name,
          arguments: item.input,
        });
        const result = await ah.callTool(env, session.ruId, item.name!, item.input ?? {});

        const redactions = detectRedactions(result);
        if (redactions.length) {
          bus.emit("dlp", "redaction", `⚠ DLP: ${redactions.join(", ")} in ${item.name} result`, {
            tool: item.name,
            redactions,
          });
        }
        bus.emit("agent_handler", "tool_result", `${item.name} → result`, {
          name: item.name,
          result,
        });
        session.history.push({ type: "tool_result", tool_use_id: item.id, content: result });
      }
    } else {
      const text = (output.content[0] as GatewayContentItem).text ?? "";
      bus.emit("response", "message", text, { text });
      return text;
    }
  }
}
```

```ts
export function createApiPlugin(env: Env): PluginOption {
  let session: Session = createSession(env);

  const setSession = (s: Session) => (session = s);
  const getSession = () => session;

  return {
    name: "vhol-api",
    configureServer(server) {
      server.middlewares.use("/api/chat", chatHandler(env, getSession));
      server.middlewares.use("/api/events", eventsHandler());
      // session/start + reset are registered in Task 11:
      registerSessionRoutes(server, env, getSession, setSession);
    },
  };
}

function chatHandler(env: Env, getSession: () => Session) {
  return async (req: IncomingMessage, res: ServerResponse) => {
    if (req.method !== "POST") return sendJson(res, 405, { error: "Method not allowed" });
    try {
      const { message } = await readJson<{ message?: string }>(req);
      if (!message) return sendJson(res, 400, { error: "message is required" });
      const text = await runTurn(env, getSession(), message);
      sendJson(res, 200, { text });
    } catch (err) {
      bus.emit("error", "error", String(err), { detail: String(err) });
      sendJson(res, 500, { error: String(err) });
    }
  };
}

function eventsHandler() {
  return (req: IncomingMessage, res: ServerResponse) => {
    res.writeHead(200, {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache",
      Connection: "keep-alive",
    });
    res.write("\n");
    for (const event of bus.getLog()) res.write(`data: ${JSON.stringify(event)}\n\n`);
    const unsub = bus.subscribe((event) => res.write(`data: ${JSON.stringify(event)}\n\n`));
    req.on("close", unsub);
  };
}
```

```tsx
import type { AgentEvent } from "../../lib/events";

export interface Status {
  ruId: string;
  model: string;
  tokens: number;
}

export function deriveStatus(events: AgentEvent[]): Status {
  let ruId = "—";
  let model = "—";
  let tokens = 0;
  for (const e of events) {
    const d = (e.detail ?? {}) as Record<string, any>;
    if (e.kind === "ru_create") ruId = d.response?.id ?? ruId;
    if (e.kind === "ru_exists") ruId = d.registeredUser?.id ?? ruId;
    if (e.kind === "route") {
      if (typeof d.model === "string" && d.model) model = d.model;
      if (d.usage) tokens += (d.usage.input_tokens ?? 0) + (d.usage.output_tokens ?? 0);
    }
  }
  return { ruId, model, tokens };
}

function Chip({ label, value }: { label: string; value: string }) {
  return (
    <span className="bg-neutral-800 border border-neutral-700 rounded-full px-3 py-0.5">
      {label} <b className="text-white">{value}</b>
    </span>
  );
}

export default function StatusBar({ events }: { events: AgentEvent[] }) {
  const { ruId, model, tokens } = deriveStatus(events);
  return (
    <div className="flex gap-2 flex-wrap text-[11px] text-neutral-300 px-3 py-2 border-b border-neutral-800">
      <Chip label="RU" value={ruId} />
      <Chip label="Model" value={model} />
      <Chip label="Tokens" value={String(tokens)} />
    </div>
  );
}
```

```tsx
export default function ConsolePage() {

  const pinnedRef = useRef(false);

  // auto-follow newest unless the user has pinned a row
  useEffect(() => {
```
